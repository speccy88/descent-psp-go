cmake_minimum_required(VERSION 3.16)
project(descent_go C)
set(CMAKE_C_STANDARD 99)
option(DESCENT_DIAGNOSTIC "Build hardware diagnostic" OFF)
include(engine-sources.cmake)
if(PSP)
 if(DESCENT_DIAGNOSTIC)
  add_executable(descent-go platform/psp/diagnostic.c)
 else()
  list(REMOVE_ITEM ENGINE_SOURCES engine/arch/sdl/digi_mixer_music.c)
  add_executable(descent-go ${ENGINE_SOURCES} platform/psp/music.c platform/psp/storage.c platform/psp/platform.c platform/psp/present.c platform/psp/input.c)
  set_source_files_properties(engine/main/inferno.c PROPERTIES COMPILE_DEFINITIONS "main=descent_main")
 endif()
 target_compile_definitions(descent-go PRIVATE PSP_PORT WORDS_NEED_ALIGNMENT)
 target_link_libraries(descent-go PRIVATE SDL_mixer SDL physfs vorbisfile vorbis ogg mikmod pspirkeyb pspdebug pspgu "-Wl,--start-group" m pthreadglue pthread cglue c gcc "-Wl,--end-group" psputility psprtc pspnet_inet pspnet_resolver pspsdk pspmodinfo pspge pspdisplay pspctrl pspaudio psppower psphprm pspuser)
 target_link_options(descent-go PRIVATE -nodefaultlibs "-Wl,-Map=${CMAKE_BINARY_DIR}/descent-go.map")
 execute_process(COMMAND $ENV{PSPDEV}/bin/mksfoex -d MEMSIZE=1 -d EFAWARE=1 -s APP_VER=00.01 "Descent PSP Go" ${CMAKE_BINARY_DIR}/PARAM.SFO COMMAND_ERROR_IS_FATAL ANY)
 create_pbp_file(TARGET descent-go TITLE "Descent PSP Go" VERSION 00.01 MEMSIZE 1 SFO_PATH ${CMAKE_BINARY_DIR}/PARAM.SFO)
else()
 find_package(PkgConfig REQUIRED)
 pkg_check_modules(SDL REQUIRED sdl)
 pkg_check_modules(MIX REQUIRED SDL_mixer)
 pkg_check_modules(PHYS REQUIRED physfs)
 add_executable(descent-go ${ENGINE_SOURCES})
 target_compile_definitions(descent-go PRIVATE __LINUX__ SHAREPATH=".")
 get_filename_component(SDL_PARENT "${SDL_INCLUDE_DIRS}" DIRECTORY)
 target_include_directories(descent-go PRIVATE ${SDL_PARENT} ${SDL_INCLUDE_DIRS} ${MIX_INCLUDE_DIRS} ${PHYS_INCLUDE_DIRS})
 target_link_directories(descent-go PRIVATE ${SDL_LIBRARY_DIRS} ${MIX_LIBRARY_DIRS} ${PHYS_LIBRARY_DIRS})
 foreach(dir IN LISTS MIX_LIBRARY_DIRS)
  target_link_options(descent-go PRIVATE "-Wl,-rpath-link,${dir}" "-Wl,-rpath-link,${dir}/pulseaudio")
 endforeach()
 target_link_libraries(descent-go PRIVATE ${SDL_LIBRARIES} ${MIX_LIBRARIES} ${PHYS_LIBRARIES} m)
endif()
target_include_directories(descent-go PRIVATE engine/include engine/main engine/arch/include platform/psp)
target_compile_definitions(descent-go PRIVATE DXX_VERSION_MAJORi=0 DXX_VERSION_MINORi=58 DXX_VERSION_MICROi=1 NO_ASM NETWORK USE_SDLMIXER RELEASE NDEBUG)
target_compile_options(descent-go PRIVATE -O2 -g -funsigned-char -fwrapv -Wall -Werror=implicit-function-declaration)
